# @meld-ts/core/a11y 探针与全量数据（Plan B）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用探针从真实 lint 生成全量规则表与 ARIA 数据，替换 Plan A 的手写缩小版；`@meld-ts/core/a11y` 的类型契约保持不变。

**Architecture:** `probe({ cwd, lints })` 在目标项目里用项目自己的 lint 跑 fixture，解析 JSON 输出得到 `required` / `supported` / `requiredWhenFocusable` 三张表，经自检与三条不变量后序列化为 TS 类型字面量。维护者模式由 `scripts/a11y-sync.ts` 在 `.cache/a11y-lab/` 里装最新 lint 并调用探针；用户模式（CLI）留给 Plan C。ARIA 数据（属性、取值、role、名字规则、idlist/id 分类）从 aria-query 生成。

**Tech Stack:** TypeScript 5.9.3、Bun（spawn / `bun init` / `bun add`）、Biome、aria-query 5.x、eslint + eslint-plugin-jsx-a11y、oxlint、@biomejs/biome

**Spec:** `docs/requirements/a11y-design.md`（§2、§4–§7、§9、§12）。三个 plan 的分工：

- Plan A（已完成）：类型核心 + 运行时，手写缩小数据
- **Plan B（本文）**：探针 + 全量数据生成
- Plan C：`meld a11y probe` 用户模式、一致性/反向/构建产物测试、AGENTS 规则

**执行方式：** SDD（每任务独立实现 + 审查）；修复成批进行，复审用 delta 包（本仓库链路经 Zscaler，避免每轮传全量 diff）。

## Global Constraints

- 设计依据 `docs/requirements/a11y-design.md`；§11 待定项按建议执行。
- **AriaKey 命名规则（本次裁决，覆盖 §3 示例里的可读名）：** 输入 key 就是输出形态去掉 `aria-` 前缀的小写形式——`checked`、`valuenow`、`labelledby`、`activedescendant`。不引入 keyMap、不提供 keyMap 重载。输出键仍是 `` `aria-${Lowercase<K>}` ``。
- `probe/` 只在 node/Bun 里跑；`src/a11y/index.ts` 不导入它，browser 包与 `@meld-ts/core/a11y` 不得出现 node 依赖。

- 生成文件确定性：role / 属性 / 键一律排序后输出；重复跑同一 lint 版本 `git diff` 必须为空。
- `lints/*.ts` 头部写「生成文件，勿手改」+ lint 与版本；更新单个 lint 整文件覆盖。

```ts
/** aria-query 的输入形状（只取生成需要的最小字段） */
export interface AriaPropertySource {
  readonly name: string;
  readonly type: string;
  readonly values?: ReadonlyArray<string | boolean>;
}

export interface RoleSource {
  readonly name: string;
  readonly abstract: boolean;
  readonly requiredProps: Readonly<Record<string, string | null>>;
  readonly nameFrom?: ReadonlyArray<string>;
  readonly accessibleNameRequired?: boolean;
}

export interface AriaQuerySource {
  readonly aria: ReadonlyArray<AriaPropertySource>;
  readonly roles: ReadonlyArray<RoleSource>;
}

const keyOf = (ariaName: string): string => ariaName.replace(/^aria-/, '');

const byKey = (a: string, b: string): number => (a < b ? -1 : a > b ? 1 : 0);

const unionOf = (values: ReadonlyArray<string | boolean>): string => {
  const parts = values
    .filter((v) => v !== 'undefined')
    .map((v) => (typeof v === 'string' ? `'${v}'` : String(v)));
  const hasTrue = parts.includes('true');
  const hasFalse = parts.includes('false');
  const rest = parts.filter((p) => p !== 'true' && p !== 'false');
  const head = hasTrue && hasFalse ? 'boolean' : rest.length ? [] : [];
  const all = head.concat(rest);
  return all.length ? all.join(' | ') : 'string';
};

const valueOf = (prop: AriaPropertySource): string => {
  switch (prop.type) {
    case 'boolean':
      return 'boolean';
    case 'tristate':
      return "boolean | 'mixed'";
    case 'integer':
    case 'number':
      return 'number';
    case 'id':
    case 'string':
      return 'string';
    case 'idlist':
      return 'string | readonly string[]';
    case 'token':
    case 'tokenlist':
      return unionOf(prop.values ?? []);
    default:
      return 'string';
  }
};

const sampleOf = (prop: AriaPropertySource): string => {
  switch (prop.type) {
    case 'boolean':
      return 'true';
    case 'tristate':
      return "'mixed'";
    case 'integer':
    case 'number':
      return '1';
    case 'id':
    case 'idlist':
      return "'x'";
    case 'token':
    case 'tokenlist': {
      const first = (prop.values ?? []).find(
        (v): v is string => typeof v === 'string' && v !== 'undefined',
      );
      return first ? `'${first}'` : 'true';
    }
    default:
      return "'x'";
  }
};

const nameRuleOf = (role: RoleSource): 'required' | 'optional' | 'prohibited' => {
  const from = role.nameFrom ?? [];
  if (from.includes('prohibited')) return 'prohibited';
  if (role.accessibleNameRequired === true && !from.includes('contents')) {
    return 'required';
  }
  return 'optional';
};

export const generateAriaTypes = (src: AriaQuerySource): string => {
  const props = [...src.aria].sort((a, b) => byKey(a.name, b.name));
  const roles = src.roles
    .filter((r) => !r.abstract)
    .sort((a, b) => byKey(a.name, b.name));
  const keys = props.map((p) => keyOf(p.name));

  const valueRows = props
    .map((p) => `  K extends '${keyOf(p.name)}' ? ${valueOf(p)} :`)
    .join('\n');

  const roleUnion = roles.map((r) => `  | '${r.name}'`).join('\n');
  const nameRows = roles
    .map((r) => `  readonly ${r.name}: '${nameRuleOf(r)}';`)
    .join('\n');

  const idLists = keys.filter((k) => {
    const p = props.find((x) => keyOf(x.name) === k);
    return p?.type === 'idlist';
  });
  const ids = keys.filter((k) => {
    const p = props.find((x) => keyOf(x.name) === k);
    return p?.type === 'id';
  });

  return `/**
 * ARIA 基础数据。
 *
 * 生成文件，勿手改。由 scripts/a11y-sync.ts 从 aria-query 生成。
 */

/** ARIA 属性名（输出形态去掉 aria- 前缀）。 */
export type AriaKey =
${keys.map((k) => `  | '${k}'`).join('\n')};

/** 引用多个 id 的属性（aria-query 类型 idlist）。 */
export type AriaIdListKey = ${idLists.length ? idLists.map((k) => `'${k}'`).join(' | ') : 'never'};

/** 引用单个 id 的属性（aria-query 类型 id）。 */
export type AriaIdKey = ${ids.length ? ids.map((k) => `'${k}'`).join(' | ') : 'never'};

/** 每个属性的合法取值（aria-query 生成）。 */
export type AriaValue<K extends AriaKey> =
${valueRows}
  string;

/** 可用的 role（aria-query 非抽象 role）。 */
export type A11yRole =
${roleUnion};

/** 每个 role 的名字规则。 */
export type A11yNameRule = {
${nameRows}
};
`;
};

export const generateSamples = (src: AriaQuerySource): string => {
  const props = [...src.aria].sort((a, b) => byKey(a.name, b.name));
  const rows = props
    .map((p) => `  '${keyOf(p.name)}': ${sampleOf(p)},`)
    .join('\n');
  return `/**
 * 每个 AriaKey 的合法样例值。
 *
 * 生成文件，勿手改。探针与一致性测试共用；不进入 a11y 入口。
 */
export const SAMPLES = {
${rows}
} as const;
`;
};
```

```ts
export interface ExecResult {
  readonly code: number;
  readonly stdout: string;
  readonly stderr: string;
}

export type ExecFn = (
  cmd: readonly string[],
  opts: { readonly cwd: string },
) => ExecResult;

export const nodeExec: ExecFn = (cmd, { cwd }) => {
```

`serialize.ts`：

```ts
export interface RoleRuleData {
  readonly required: readonly string[];
  readonly supported: readonly string[];
  readonly requiredWhenFocusable: readonly string[];
}

export interface ProfileData {
  readonly lint: string;
  readonly version: string;
  readonly aliasOf?: string;
  readonly roles: Readonly<Record<string, RoleRuleData>>;
}

const SORT = (a: string, b: string): number => (a < b ? -1 : a > b ? 1 : 0);

const keyOf = (ariaName: string): string => ariaName.replace(/^aria-/, '');

const union = (names: readonly string[]): string =>
  names.length ? names.map((n) => `'${keyOf(n)}'`).join(' | ') : 'never';

const sorted = (names: readonly string[]): string[] =>
  [...new Set(names)].sort(SORT);

const typeName = (lint: string): string =>
  `${lint[0].toUpperCase()}${lint.slice(1)}A11yProfile`;

export const serializeProfile = (profile: ProfileData): string => {
  const alias = profile.aliasOf ? `; aliasOf: '${profile.aliasOf}'` : '';
  const roles = Object.keys(profile.roles)
    .sort(SORT)
    .map((role) => {
      const r = profile.roles[role]!;
      return `    ${role}: { required: ${union(sorted(r.required))}; supported: ${union(sorted(r.supported))}; requiredWhenFocusable: ${union(sorted(r.requiredWhenFocusable))} };`;
    })
    .join('\n');
  return `/**
 * ${profile.lint} 内置规则表。
 *
 * 生成文件，勿手改。${profile.lint} ${profile.version}。
 */
export type ${typeName(profile.lint)} = {
  meta: { lint: '${profile.lint}'; version: '${profile.version}'; source: 'builtin'${alias} };
  roles: {
${roles}
  };
};
`;
};
```

```ts
// 命令：bun lint --no-script -f json \
```

```ts
import type { ExecFn } from '../exec';
import type { Fixture } from '../fixtures';

export interface RunnerOptions {
  readonly bin: readonly string[];
  readonly cwd: string;
  readonly workDir: string;
  readonly writeFile: (path: string, content: string) => void;
  readonly exec: ExecFn;
}

export interface LintReport {
  readonly missing: ReadonlyArray<{ role: string; props: string[] }>;
  readonly unsupported: ReadonlyArray<{ role: string; prop: string }>;
}

const REQ =
  /role "([^"]+)" must have the following attributes defined: (.+)$/;
const SUP = /The attribute (aria-[a-z]+) is not supported by the role ([a-z]+)/;

export const parseEslint = (raw: string): LintReport => {
  const missing: Array<{ role: string; props: string[] }> = [];
  const unsupported: Array<{ role: string; prop: string }> = [];
  const files = JSON.parse(raw) as Array<{
    messages: Array<{ ruleId: string | null; message: string }>;
  }>;
  for (const file of files) {
    for (const m of file.messages) {
      if (m.ruleId !== 'jsx-a11y/role-has-required-aria-props' &&
          m.ruleId !== 'jsx-a11y/role-supports-aria-props') {
        continue;
      }
      const req = m.message.match(REQ);
      if (req) {
        missing.push({
          role: req[1]!,
          props: req[2]!.split(',').map((s) => s.trim()),
        });
        continue;
      }
      const sup = m.message.match(SUP);
      if (sup) unsupported.push({ role: sup[2]!, prop: sup[1]! });
    }
  }
  return { missing, unsupported };
};

const CONFIG = `import jsxA11y from 'eslint-plugin-jsx-a11y';
export default [{
  files: ['**/*.jsx'],
  plugins: { 'jsx-a11y': jsxA11y },
  languageOptions: {
    parserOptions: {
      ecmaVersion: 'latest',
      sourceType: 'module',
      ecmaFeatures: { jsx: true },
    },
  },
  rules: {
    'jsx-a11y/role-has-required-aria-props': 'error',
    'jsx-a11y/role-supports-aria-props': 'error',
  },
}];
`;

export const runEslint = (
  fixtures: readonly Fixture[],
  { bin, cwd, workDir, writeFile, exec }: RunnerOptions,
): LintReport => {
  writeFile(`${workDir}/config/eslint/eslint.config.js`, CONFIG);
  for (const f of fixtures) writeFile(`${workDir}/fixtures/${f.name}.jsx`, f.code);
  const r = exec(
    [...bin, '-c', `${workDir}/config/eslint/eslint.config.js`, '-f', 'json',
      `${workDir}/fixtures`],
    { cwd },
  );
  return parseEslint(r.stdout);
};
```

实现要点：

- required 循环：跑 → 解析缺失 → 补上 → 再跑，直到无缺失或达到 10 轮上限；超过上限视为解析失败。
- supported：初始集合 = aria-query 的属性全集，减去该 lint 报「不支持」的；再减去 aria-query `prohibited` 属性（传入 `nameRule` 时一并减 label/labelledby）。注意：supported 的判定用「没报不支持」而不是「报支持」。
- requiredWhenFocusable = focusableRequired − required。
- 自检：对 switch/checkbox/slider 中至少一个，故意少写一个必填属性，确认 runner 报错；否则抛错。
- 不变量：三条逐 role 断言，失败抛错。
- 单测：注入假 runner 与假 fs（writeFile/mkdir/rm/exists），断言调用序列与最终 ProfileData；解析器细节在 runner 单测。

`probe` 的核心实现：

```ts
import type { ExecFn } from './exec';
import type { ProfileData } from './serialize';

export interface ProbeFs {
  readonly mkdir: (path: string) => void;
  readonly write: (path: string, content: string) => void;
  readonly rm: (path: string) => void;
  readonly exists: (path: string) => boolean;
}

export interface ProbeInput {
  readonly cwd: string;
  readonly lints: ReadonlyArray<'eslint' | 'oxlint' | 'biome' | 'bun'>;
  readonly samples: Readonly<Record<string, string>>;
  readonly keys: readonly string[];
  readonly nameRule: Readonly<Record<string, 'required' | 'optional' | 'prohibited'>>;
  readonly exec: ExecFn;
  readonly fs: ProbeFs;
  readonly run: (lint: string, workDir: string) => {
    missing: ReadonlyArray<{ role: string; props: string[] }>;
    unsupported: ReadonlyArray<{ role: string; prop: string }>;
  };
  readonly versions: Readonly<Record<string, string>>;
  readonly keep?: boolean;
}

const MAX_ROUNDS = 10;

export const probe = (input: ProbeInput): Record<string, ProfileData> => {
  const workDir = `${input.cwd}/.meld-a11y-probe`;
  const out: Record<string, ProfileData> = {};
  try {
    input.fs.rm(workDir);
    input.fs.mkdir(`${workDir}/fixtures`);
    for (const lint of input.lints) {
      const required: Record<string, string[]> = {};
      // required：循环补属性直到 lint 不再报缺失
      const pending = new Set<string>();
      for (let round = 0; round < MAX_ROUNDS; round++) {
        const report = input.run(lint, workDir);
        let changed = false;
        for (const item of report.missing) {
          if (pending.has(item.role)) continue;
          required[item.role] = [...(required[item.role] ?? []), ...item.props];
          pending.add(item.role);
          changed = true;
        }
        if (!changed) break;
        if (round === MAX_ROUNDS - 1) {
          throw new Error(`${lint}: required probing did not converge`);
        }
      }
      const report = input.run(lint, workDir);
      const unsupportedByRole = new Map<string, Set<string>>();
      for (const s of report.unsupported) {
        if (!unsupportedByRole.has(s.role)) unsupportedByRole.set(s.role, new Set());
        unsupportedByRole.get(s.role)!.add(s.prop);
      }
      const roles: ProfileData['roles'] = {};
      for (const role of Object.keys(input.nameRule).sort()) {
        const req = (required[role] ?? []).filter(
          (p) => input.nameRule[role] !== 'prohibited' ||
            (p !== 'aria-label' && p !== 'aria-labelledby'),
        );
        const banned = unsupportedByRole.get(role) ?? new Set<string>();
        const supported = input.keys
          .map((k) => `aria-${k}`)
          .filter((p) => !banned.has(p))
          .filter((p) => !(input.nameRule[role] === 'prohibited' &&
            (p === 'aria-label' || p === 'aria-labelledby')));
        roles[role] = {
          required: req,
          supported,
          requiredWhenFocusable: [], // focusable 差异由 run 的 focusable 变体补齐
        };
      }
      out[lint] = { lint, version: input.versions[lint]!, roles };
    }
    return out;
  } finally {
    if (!input.keep) input.fs.rm(workDir);
  }
};
```

真实实现还要：把 `run` 拆成 required / supported / focusable 三次调用（或让 `run` 接受模式参数），focusable 表取两次 required 的差集；自检与不变量在写文件前调用（`assertInvariants`、`selfCheck`）。单元测试覆盖：不收敛时抛错、union 覆盖被减、prohibited role 的 label/labelledby 被减、focusable 差集、keep=false 时清理。
