# @meld-ts/core/a11y 核心类型与运行时（Plan A）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 `@meld-ts/core/a11y` 子路径的类型机器与 `createA11y` 运行时——按 lint 规则表在类型层约束 aria 属性输入，输出类型精确的属性对象；不导出任何预绑定实例，只经 `package.json` exports 子路径提供 `createA11y`。

**Architecture:** 类型层以 `A11yProfile` 字面量规则表（每 lint 一张）约束输入并推断输出键值；运行时只做值转换（html 字符串化、idlist 空格拼接、undefined 跳过、focusable → tabIndex）。内置表声明为 interface，留给使用方以 declaration merging 重载；多 lint 在类型层按「必填并集 / 允许交集 / 可聚焦必填并集」合并。本 plan 用手写缩小版数据（4 role × biome/oxlint 两张表）打通全链路；Plan B 由探针生成全量数据、整文件覆盖。

**Tech Stack:** TypeScript 5.9.3（仓库锁定版本；附录 A 已在 TS 7.0.2 与 5.9.3 双版本实测）、bun test、bunup、Biome 2.x

**Spec:** `docs/requirements/a11y-design.md`（§1–§5、§9、§11、附录 A）。spec 拆为三个 plan：

- **Plan A（本文）**：类型核心 + 运行时，数据为手写缩小版（biome/oxlint × 4 role）
- Plan B：探针 + `a11y:sync` 生成全量数据（`aria.generated.ts`、四张 lint 表、`samples.generated.ts`）
- Plan C：`meld a11y probe` bin、用户模式重载文件、一致性/反向/构建产物测试、AGENTS 规则

## Global Constraints

- 设计依据 `docs/requirements/a11y-design.md`；§11 四条待定按「建议」列执行：命令形态 `meld a11y probe`、输入 camelCase、**不导出默认 a11y（只经 exports 子路径提供 `createA11y`）**、bun 表在 Plan B 以 `aliasOf: 'eslint'` 补齐。
- 所有工作在 worktree 分支 `feat/a11y`，每个 Task 一个 commit（conventional commits）。
- TDD：先写测试/类型断言 → 验证红 → 最小实现 → 验证绿 → commit。类型断言的红绿以 `bun run ts-check`（tsc 5.9.3）为准；`bun test` 单跑不做类型检查。
- 代码规范：Biome（2 空格、单引号、分号、80 列、LF）；`// biome-ignore` 必带原因；注释只写 WHY；JSDoc 含 `@example`。
- `src/browser/index.ts` 合并所有子路径导出：导出名统一 `A11y`/`Aria` 前缀，不得与现有导出（types/guards/base/timer/singleton/path/traits/events/async）撞名。
- `aria.generated.ts` 与 `lints/*.ts` **只含类型**（无运行时值）；Plan A 为手写缩小版，文件头注明 Plan B 整文件覆盖。

```ts
/**
 * ARIA 基础数据。
 *
 * 从 aria-query 生成。Plan A 阶段为手写缩小版（4 role、部分属性），
 * Plan B（bun run a11y:sync）整文件覆盖为全量生成版。
 */

/** camelCase 的 ARIA 属性名（a11y() 的输入属性名）。 */
export type AriaKey =
  | 'checked'
  | 'expanded'
  | 'controls'
  | 'label'
  | 'labelledBy'
  | 'valueNow'
  | 'selected'
  | 'details'
  | 'hidden';

/**
 * 引用多个 id 的属性（aria-query 类型 idlist）：
 * 接受 string 或 readonly string[]，输出时空格拼接。
 */
export type AriaIdListKey = 'controls' | 'labelledBy';

/** 引用单个 id 的属性（aria-query 类型 id）：只接受单个 string。 */
export type AriaIdKey = 'details';

/** 每个属性的合法取值（aria-query 生成）。 */
export type AriaValue<K extends AriaKey> = K extends 'checked'
  ? boolean | 'mixed'
  : K extends 'expanded' | 'selected' | 'hidden'
    ? boolean
    : K extends 'valueNow'
      ? number
      : K extends AriaIdListKey
        ? string | readonly string[]
        : K extends AriaIdKey
          ? string
          : string;

/** 可用的 role（aria-query 非抽象 role 的缩小集）。 */
export type A11yRole = 'switch' | 'img' | 'separator' | 'combobox';

/**
 * 每个 role 的名字规则（由 aria-query 的 accessibleNameRequired /
 * nameFrom 归出三类）：
 * - required：作者必须给名
 * - optional：名字可由子元素内容提供
 * - prohibited：不允许给名
 */
export type A11yNameRule = {
  readonly switch: 'required';
  readonly img: 'required';
  readonly separator: 'optional';
  readonly combobox: 'required';
};
```

```ts
import type { A11yRole, AriaKey } from './aria.generated';
import type { A11yBuiltinLints } from './lints';

/**
 * 单个 role 的一张规则条目：
 * 必填 / 允许（含必填，已减去禁止属性）/ 可聚焦时额外必填。

export type A11yRoleRule = {
  readonly required: AriaKey;
  readonly supported: AriaKey;
  readonly requiredWhenFocusable: AriaKey;
};

/** 一张 lint 规则表：某个 lint、某个版本下的规则。由探针生成，不手写。 */
export type A11yProfile = {
  readonly meta: {
    readonly lint: string;
    readonly version: string;
    readonly source: 'builtin' | 'probe' | 'merged';
    /** 这张表直接沿用另一个 lint 的表（bun 表 aliasOf eslint） */
    readonly aliasOf?: string;
  };
  readonly roles: { readonly [R in A11yRole]: A11yRoleRule };
};

/** 生效的 lint 名字（内置表的键）。 */
export type A11yLintName = keyof A11yBuiltinLints;
```

```ts
// ---- 多 lint 合并 ----

type UnionToIntersection<U> = (
  U extends unknown ? (x: U) => void : never
) extends (x: infer I) => void
  ? I
  : never;

type Intersect<Ps extends A11yProfile, R extends A11yRole> = Extract<
  AriaKey,
  UnionToIntersection<
    Ps extends unknown ? (k: Ps['roles'][R]['supported']) => void : never
  > extends infer F
    ? F extends (k: infer K) => void
      ? K
      : never
    : never
>;

/**
 * 合并多张规则表：
 * - required 取并集（每个 lint 都要求 → 合并后都要求）
 * - supported 取交集（每个 lint 都允许 → 合并后才允许）
 * - requiredWhenFocusable 取并集
 */
export type MergeProfiles<Ps extends A11yProfile> = {
  meta: {
    lint: Ps['meta']['lint'];
    version: Ps['meta']['version'];
    source: 'merged';
  };
  roles: {
    [R in A11yRole]: {
      required: Ps['roles'][R]['required'];
      supported: Intersect<Ps, R>;
      requiredWhenFocusable: Ps['roles'][R]['requiredWhenFocusable'];
    };
  };
};

export type ResolveProfile<L> = L extends A11yLintName
  ? A11yLints[L]
  : L extends readonly A11yLintName[]
    ? MergeProfiles<A11yLints[L[number]]>
    : never;

// ---- 输入 ----

type Rule<P extends A11yProfile, R extends A11yRole> = P['roles'][R];

type NameRequirement<N> = N extends 'required'
  ? { label: string } | { labelledBy: string | readonly string[] }
  : {};

export type A11yInput<P extends A11yProfile, R extends A11yRole> =
  & { [K in Rule<P, R>['required']]-?: AriaValue<K> }
  & {
      [K in Exclude<
        Rule<P, R>['supported'],
        Rule<P, R>['required']
      >]?: AriaValue<K> | undefined;
    }
  & { focusable?: true | undefined }
  & NameRequirement<A11yNameRule[R]>;

/**
 * 多余键检查：I 是推导出的泛型，TypeScript 对它不做多余属性检查，
 * 必须显式把允许集合外的键映射成 never。
 */
type Exact<I, Allowed extends PropertyKey> = {
  [K in Exclude<keyof I, Allowed>]: never;
};

/**
 * 可聚焦时的额外必填。
 * 条件必须是 `true extends I['focusable']`：变量类型含 true（如
 * true | undefined）也要打开额外必填。用 `I extends { focusable: true }`
 * 会判不中含 undefined 的变量。
 */
type FocusRequirement<P extends A11yProfile, R extends A11yRole, I> =
  I extends { focusable: unknown }
    ? true extends I['focusable']
      ? { [K in Rule<P, R>['requiredWhenFocusable']]-?: AriaValue<K> }
      : unknown
    : unknown;

// ---- 输出 ----

type Str<V> = V extends readonly string[]
  ? string
  : V extends boolean | number
    ? `${V}`
    : V;

type OutValue<V, O extends A11yOutputMode> = V extends readonly string[]
  ? string
  : O extends 'html'
    ? Str<V>
    : V;

type AriaAttrs<I, O extends A11yOutputMode> = {
  readonly [K in keyof I as K extends string
    ? undefined extends I[K]
      ? never
      : `aria-${Lowercase<K>}`
    : never]: OutValue<I[K], O>;
} & {
  readonly [K in keyof I as K extends string
    ? undefined extends I[K]
      ? [I[K]] extends [undefined]
        ? never
        : `aria-${Lowercase<K>}`
      : never
    : never]?: OutValue<Exclude<I[K], undefined>, O>;
};

type TabKey<O extends A11yOutputMode> = O extends 'html'
  ? 'tabindex'
  : 'tabIndex';

type TabVal<O extends A11yOutputMode> = O extends 'html' ? '0' : 0;

type FocusAttr<I, O extends A11yOutputMode> = I extends {
  focusable: unknown;
}
  ? true extends I['focusable']
    ? undefined extends I['focusable']
      ? { readonly [K in TabKey<O>]?: TabVal<O> }
      : { readonly [K in TabKey<O>]: TabVal<O> }
    : {}
  : {};

export type A11yAttrs<R, I, O extends A11yOutputMode> = {
  readonly role: R;
} & FocusAttr<I, O> &
  AriaAttrs<Omit<I, 'focusable'>, O>;

/**
 * a11y 可调用对象：
 * - a11y(role, props)：按 role 约束输入，输出精确属性
 * - a11y.aria(props)：原生元素补状态，不带 role
 */
export type A11y<P extends A11yProfile, O extends A11yOutputMode> = {
  <R extends A11yRole, const I extends A11yInput<P, R>>(
    role: R,
    props: I & Exact<I, Rule<P, R>['supported'] | 'focusable'> &
      FocusRequirement<P, R, I>,
  ): A11yAttrs<R, I, O>;
  aria<const I extends { [K in AriaKey]?: AriaValue<K> | undefined }>(
    props: I & Exact<I, AriaKey>,
  ): AriaAttrs<I, O>;
};

// ---- 选项解析 ----

/** output 没写时取 'react' */
type OutputOf<Opt extends A11yOptions> = Opt['output'] extends A11yOutputMode
  ? Opt['output']
  : 'react';

type AsProfile<P> = P extends A11yProfile ? P : never;

export type A11yOf<Opt extends A11yOptions> = A11y<
  AsProfile<ResolveProfile<Opt['lint']>>,
  OutputOf<Opt>
>;
```

```ts
import type { AriaIdListKey } from './aria.generated';
import type { A11yOf, A11yOptions, OverrideKeyCheck } from './types';

/**
 * idlist 属性的运行时最小表：只有这些键的数组值用空格拼接。
 * 其余逻辑全部由类型层保证。
 */
const IDLIST_KEYS = ['controls', 'labelledBy'] as const;
const IDLIST_SET: ReadonlySet<string> = new Set(IDLIST_KEYS);

type Eq<A, B> = (<T>() => T extends A ? 1 : 2) extends <T>() => T extends B
  ? 1
  : 2
  ? true
  : false;
type Assert<T extends true> = T;
/** 生成数据更新而此处忘记同步时，编译期报错 */
type IdListInSync = Assert<Eq<(typeof IDLIST_KEYS)[number], AriaIdListKey>>;

/** 实现内部用的宽松签名，对外不可见 */
interface A11yCallable {
  (role: string, props: Record<string, unknown>): Record<string, unknown>;
  aria(props: Record<string, unknown>): Record<string, unknown>;
}

export function createA11y<const Opt extends A11yOptions>(
  options: Opt & OverrideKeyCheck,
): A11yOf<Opt>;
export function createA11y(options: A11yOptions): A11yCallable {
  const html: boolean = options.output === 'html';

  const convert = (key: string, value: unknown): unknown => {
    if (IDLIST_SET.has(key) && Array.isArray(value)) {
      return value.join(' ');
    }
    if (html && (typeof value === 'boolean' || typeof value === 'number')) {
      return String(value);
    }
    return value;
  };

  const toAttrs = (props: Record<string, unknown>): Record<string, unknown> => {
    const out: Record<string, unknown> = {};
    for (const key of Object.keys(props)) {
      if (key === 'focusable') continue;
      const value = props[key];
      if (value === undefined) continue;
      out[`aria-${key.toLowerCase()}`] = convert(key, value);
    }
    return out;
  };

  // 运行时实现只能用宽松签名；对外类型由公开重载 createA11y 保证
  const a11y = ((role: string, props: Record<string, unknown>) => {
    const attrs = toAttrs(props);
    if (props.focusable === true) {
      attrs[html ? 'tabindex' : 'tabIndex'] = html ? '0' : 0;
    }
    return { role, ...attrs };
  }) as A11yCallable;

  a11y.aria = toAttrs;
  return a11y;
}
```
